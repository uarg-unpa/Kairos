USE kairos;

-- 🧱 1) Crear el proyecto si no existe
INSERT INTO proyecto (nombre, equipo, fecha_creacion, estado, fecha_actualizacion)
SELECT 'Proyecto RUP Demo', 'Equipo Kairos', CURDATE(), 'Activo', NOW()
WHERE NOT EXISTS (
  SELECT 1 FROM proyecto WHERE nombre = 'Proyecto RUP Demo'
);

-- Obtener el ID del proyecto
SELECT id_proyecto INTO @proy
FROM proyecto
WHERE nombre = 'Proyecto RUP Demo'
ORDER BY id_proyecto DESC
LIMIT 1;

-- 🧩 2) Crear Etapas solo si no existen

-- Etapa Inicio
INSERT INTO etapa (nombre, fecha_inicio, fecha_fin, id_proyecto)
SELECT 'Inicio', '2025-08-19', '2025-09-09', @proy
WHERE NOT EXISTS (
  SELECT 1 FROM etapa WHERE nombre = 'Inicio' AND id_proyecto = @proy
);

-- Etapa Elaboración
INSERT INTO etapa (nombre, fecha_inicio, fecha_fin, id_proyecto)
SELECT 'Elaboración', '2025-09-10', '2025-10-10', @proy
WHERE NOT EXISTS (
  SELECT 1 FROM etapa WHERE nombre = 'Elaboración' AND id_proyecto = @proy
);

-- Etapa Construcción
INSERT INTO etapa (nombre, fecha_inicio, fecha_fin, id_proyecto)
SELECT 'Construcción', '2025-10-11', '2025-11-14', @proy
WHERE NOT EXISTS (
  SELECT 1 FROM etapa WHERE nombre = 'Construcción' AND id_proyecto = @proy
);

-- Obtener IDs de cada etapa
SELECT id_etapa INTO @etapa_inicio
FROM etapa WHERE nombre = 'Inicio' AND id_proyecto = @proy LIMIT 1;

SELECT id_etapa INTO @etapa_elab
FROM etapa WHERE nombre = 'Elaboración' AND id_proyecto = @proy LIMIT 1;

SELECT id_etapa INTO @etapa_cons
FROM etapa WHERE nombre = 'Construcción' AND id_proyecto = @proy LIMIT 1;

-- 🧩 3) Iteraciones (solo si no existen)

-- Elaboración - Iteración 1
INSERT INTO iteracion (numero, descripcion, fecha_inicio, fecha_fin, id_etapa)
SELECT 1, 'Elaboración - Iteración 1', '2025-09-10', '2025-09-23', @etapa_elab
WHERE NOT EXISTS (
  SELECT 1 FROM iteracion WHERE numero = 1 AND id_etapa = @etapa_elab
);

-- Elaboración - Iteración 2
INSERT INTO iteracion (numero, descripcion, fecha_inicio, fecha_fin, id_etapa)
SELECT 2, 'Elaboración - Iteración 2', '2025-09-24', '2025-10-10', @etapa_elab
WHERE NOT EXISTS (
  SELECT 1 FROM iteracion WHERE numero = 2 AND id_etapa = @etapa_elab
);

-- Construcción - Iteración 1
INSERT INTO iteracion (numero, descripcion, fecha_inicio, fecha_fin, id_etapa)
SELECT 1, 'Construcción - Iteración 1', '2025-10-11', '2025-10-29', @etapa_cons
WHERE NOT EXISTS (
  SELECT 1 FROM iteracion WHERE numero = 1 AND id_etapa = @etapa_cons
);

-- Construcción - Iteración 2
INSERT INTO iteracion (numero, descripcion, fecha_inicio, fecha_fin, id_etapa)
SELECT 2, 'Construcción - Iteración 2', '2025-10-29', '2025-11-07', @etapa_cons
WHERE NOT EXISTS (
  SELECT 1 FROM iteracion WHERE numero = 2 AND id_etapa = @etapa_cons
);

-- Construcción - Iteración 3
INSERT INTO iteracion (numero, descripcion, fecha_inicio, fecha_fin, id_etapa)
SELECT 3, 'Construcción - Iteración 3', '2025-11-08', '2025-11-14', @etapa_cons
